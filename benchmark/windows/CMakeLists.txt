if(NOT WIN32)
    return()
endif()
set(FAH_WEBVIEW2_SDK "" CACHE PATH "Extracted Microsoft.Web.WebView2 NuGet SDK for the optional comparison host")
set(FAH_WEBVIEW2_SUPPORT_INCLUDE "" CACHE PATH "Additional Windows SDK header directory for cross-building")
if(NOT FAH_WEBVIEW2_SDK)
    message(STATUS "WebView2 comparison host disabled: set FAH_WEBVIEW2_SDK to an extracted SDK")
    return()
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR CMAKE_SYSTEM_PROCESSOR MATCHES "[Aa][Rr][Mm]|aarch64" OR CMAKE_GENERATOR_PLATFORM MATCHES "[Aa][Rr][Mm]")
    message(FATAL_ERROR "The initial WebView2 comparison bundle requires an x64 Windows target")
endif()
find_path(FAH_WEBVIEW2_INCLUDE WebView2.h PATHS "${FAH_WEBVIEW2_SDK}/build/native/include" NO_DEFAULT_PATH REQUIRED)
find_file(FAH_WEBVIEW2_LOADER WebView2Loader.dll PATHS "${FAH_WEBVIEW2_SDK}/build/native/x64" NO_DEFAULT_PATH REQUIRED)

add_executable(fah_webview_benchmark WIN32 WebViewBenchmark.cpp)
set_target_properties(fah_webview_benchmark PROPERTIES OUTPUT_NAME fah-webview-benchmark)
target_compile_features(fah_webview_benchmark PRIVATE cxx_std_17)
target_compile_definitions(fah_webview_benchmark PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0a00)
target_include_directories(fah_webview_benchmark PRIVATE "${PROJECT_SOURCE_DIR}/third_party")
target_include_directories(fah_webview_benchmark SYSTEM PRIVATE "${FAH_WEBVIEW2_INCLUDE}")
if(FAH_WEBVIEW2_SUPPORT_INCLUDE)
    target_include_directories(fah_webview_benchmark SYSTEM PRIVATE "${FAH_WEBVIEW2_SUPPORT_INCLUDE}")
endif()
target_link_libraries(fah_webview_benchmark PRIVATE fah_core user32 gdi32 shell32 shlwapi ole32 uuid psapi)
if(MSVC)
    target_compile_options(fah_webview_benchmark PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(fah_webview_benchmark PRIVATE -Wall -Wextra)
    target_link_options(fah_webview_benchmark PRIVATE -municode -Wl,--subsystem,windows)
endif()
add_custom_command(TARGET fah_webview_benchmark POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${FAH_WEBVIEW2_LOADER}" "$<TARGET_FILE_DIR:fah_webview_benchmark>/WebView2Loader.dll"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${FAH_WEBVIEW2_SDK}/LICENSE.txt" "$<TARGET_FILE_DIR:fah_webview_benchmark>/WEBVIEW2-SDK-LICENSE.txt"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${PROJECT_SOURCE_DIR}/benchmark/web" "$<TARGET_FILE_DIR:fah_webview_benchmark>/web")
