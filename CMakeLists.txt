cmake_minimum_required(VERSION 3.21)
if(CMAKE_HOST_APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS deployment target")
endif()

project(FAHSaver VERSION 0.1.9 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(fah_core src/core/Model.cpp)
target_include_directories(fah_core PUBLIC include PRIVATE third_party)
if(MSVC)
    target_compile_options(fah_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(fah_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

find_package(Threads REQUIRED)
target_sources(fah_core PRIVATE src/core/SnapshotCache.cpp)
target_link_libraries(fah_core PUBLIC Threads::Threads)
add_library(fah_source src/net/Session.cpp)
target_include_directories(fah_source PRIVATE third_party)
target_link_libraries(fah_source PUBLIC fah_core Threads::Threads)

include(CTest)
if(BUILD_TESTING)
    add_executable(fah_core_tests tests/core_tests.cpp)
    target_link_libraries(fah_core_tests PRIVATE fah_core)
    add_test(NAME core COMMAND fah_core_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/synthetic-small.json")
    add_executable(fah_snapshot_cache_tests tests/snapshot_cache_tests.cpp)
    target_link_libraries(fah_snapshot_cache_tests PRIVATE fah_core)
    add_test(NAME snapshot_cache COMMAND fah_snapshot_cache_tests)
    set_tests_properties(snapshot_cache PROPERTIES TIMEOUT 30)
    add_executable(fah_live_model_tests tests/live_model_tests.cpp)
    target_link_libraries(fah_live_model_tests PRIVATE fah_core)
    add_test(NAME live_model COMMAND fah_live_model_tests)
    add_executable(fah_decimal_input_tests tests/decimal_input_tests.cpp)
    target_include_directories(fah_decimal_input_tests PRIVATE include)
    add_test(NAME decimal_input COMMAND fah_decimal_input_tests)
    add_executable(fah_source_tests tests/source_tests.cpp)
    target_include_directories(fah_source_tests PRIVATE third_party)
    target_link_libraries(fah_source_tests PRIVATE fah_source)
    add_test(NAME source_protocol COMMAND fah_source_tests)
    add_executable(fah_thread_sample tools/windows-thread-sample.cpp)
    target_include_directories(fah_thread_sample PRIVATE third_party)
    target_link_libraries(fah_thread_sample PRIVATE Threads::Threads)
    if(WIN32)
        target_compile_definitions(fah_thread_sample PRIVATE UNICODE _UNICODE)
        target_link_libraries(fah_thread_sample PRIVATE psapi)
        if(NOT MSVC)
            target_link_options(fah_thread_sample PRIVATE -municode)
        endif()
    endif()
    add_test(NAME thread_attribution_logic COMMAND fah_thread_sample --self-check)
endif()

if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 17)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    target_sources(fah_source PRIVATE src/net/MacSource.mm)
    set_source_files_properties(src/net/MacSource.mm PROPERTIES COMPILE_OPTIONS -fobjc-arc)
    target_link_libraries(fah_source PRIVATE "-framework Foundation" "-framework CFNetwork")
    add_executable(fah_capture tools/capture-main.mm)
    target_compile_options(fah_capture PRIVATE -fobjc-arc)
    target_link_libraries(fah_capture PRIVATE fah_source "-framework Foundation")
    add_executable(fah_web_benchmark benchmark/macos/WKBenchmark.mm)
    target_compile_options(fah_web_benchmark PRIVATE -fobjc-arc -Wall -Wextra)
    target_link_libraries(fah_web_benchmark PRIVATE "-framework AppKit" "-framework WebKit")
    add_subdirectory(platform/macos)
    add_subdirectory(benchmark/macos)
    if(BUILD_TESTING)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME mac_source_diagnostics COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/mac_source_diagnostics.py" "$<TARGET_FILE:fah_capture>")
        set_tests_properties(mac_source_diagnostics PROPERTIES TIMEOUT 20)
        add_test(NAME mac_preview_cli COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/platform/macos/test_cli.py" "$<TARGET_FILE:fah_macos_preview>")
        add_test(NAME web_benchmark_help COMMAND fah_web_benchmark --help)
        add_test(NAME web_benchmark_rejects_nan COMMAND fah_web_benchmark
            --capture absent --html absent --fps nan)
        set_tests_properties(web_benchmark_rejects_nan PROPERTIES WILL_FAIL TRUE)
    endif()
endif()

if(WIN32)
    target_sources(fah_source PRIVATE src/net/WindowsSource.cpp)
    target_compile_definitions(fah_source PRIVATE _WIN32_WINNT=0x0602)
    target_link_libraries(fah_source PRIVATE winhttp)
    add_subdirectory(platform/windows)
    if(BUILD_TESTING)
        add_executable(fah_windows_source_cancel tests/windows_source_cancel.cpp)
        target_link_libraries(fah_windows_source_cancel PRIVATE fah_source)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME windows_source_cancellation COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/windows_cancel_smoke.py" "$<TARGET_FILE:fah_windows_source_cancel>")
        set_tests_properties(windows_source_cancellation PROPERTIES TIMEOUT 90)
    endif()
endif()

add_subdirectory(benchmark/windows)
