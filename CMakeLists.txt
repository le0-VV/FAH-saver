cmake_minimum_required(VERSION 3.21)
project(FAHSaver VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(fah_core src/core/Model.cpp)
target_include_directories(fah_core PUBLIC include PRIVATE third_party)
if(MSVC)
    target_compile_options(fah_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(fah_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

find_package(Threads REQUIRED)
add_library(fah_source src/net/Session.cpp)
target_include_directories(fah_source PRIVATE third_party)
target_link_libraries(fah_source PUBLIC fah_core Threads::Threads)

include(CTest)
if(BUILD_TESTING)
    add_executable(fah_core_tests tests/core_tests.cpp)
    target_link_libraries(fah_core_tests PRIVATE fah_core)
    add_test(NAME core COMMAND fah_core_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/synthetic-small.json")
    add_executable(fah_source_tests tests/source_tests.cpp)
    target_include_directories(fah_source_tests PRIVATE third_party)
    target_link_libraries(fah_source_tests PRIVATE fah_source)
    add_test(NAME source_protocol COMMAND fah_source_tests)
endif()

if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 17)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    target_sources(fah_source PRIVATE src/net/MacSource.mm)
    set_source_files_properties(src/net/MacSource.mm PROPERTIES COMPILE_OPTIONS -fobjc-arc)
    target_link_libraries(fah_source PRIVATE "-framework Foundation")
    add_executable(fah_capture tools/capture-main.mm)
    target_compile_options(fah_capture PRIVATE -fobjc-arc)
    target_link_libraries(fah_capture PRIVATE fah_source "-framework Foundation")
    add_executable(fah_web_benchmark benchmark/macos/WKBenchmark.mm)
    target_compile_options(fah_web_benchmark PRIVATE -fobjc-arc -Wall -Wextra)
    target_link_libraries(fah_web_benchmark PRIVATE "-framework AppKit" "-framework WebKit")
    add_subdirectory(platform/macos)
    if(BUILD_TESTING)
        add_test(NAME web_benchmark_help COMMAND fah_web_benchmark --help)
        add_test(NAME web_benchmark_rejects_nan COMMAND fah_web_benchmark
            --capture absent --html absent --fps nan)
        set_tests_properties(web_benchmark_rejects_nan PROPERTIES WILL_FAIL TRUE)
    endif()
endif()

if(WIN32)
    target_sources(fah_source PRIVATE src/net/WindowsSource.cpp)
    target_compile_definitions(fah_source PRIVATE _WIN32_WINNT=0x0602)
    target_link_libraries(fah_source PRIVATE winhttp)
    add_subdirectory(platform/windows)
endif()
